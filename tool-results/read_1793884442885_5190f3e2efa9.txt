   574→    function uint8ToBase64(u8) {
   575→      let binary = '';
   576→      const chunkSize = 0x8000;
   577→
   578→      for (let i = 0; i < u8.length; i += chunkSize) {
   579→        const chunk = u8.subarray(i, i + chunkSize);
   580→        binary += String.fromCharCode.apply(null, chunk);
   581→      }
   582→
   583→      return btoa(binary);
   584→    }

   653→    function writeHashParams(paramsObject) {
   654→      const params = readHashParams();
   655→
   656→      for (const [key, value] of Object.entries(paramsObject)) {
   657→        if (value === null || value === undefined || value === '') {
   658→          params.delete(key);
   659→        } else {
   660→          params.set(key, value);
   661→        }
   662→      }
   663→
   664→      const hash = params.toString();
   665→      const newUrl = `${location.pathname}${location.search}${hash ? `#${hash}` : ''}`;
   666→
   667→      try {
   668→        history.replaceState(null, '', newUrl);
   669→      } catch {
   670→        suppressHashChange = true;
   671→        location.hash = hash;
   672→        setTimeout(() => {
   673→          suppressHashChange = false;
   674→        }, 0);
   675→      }
   676→    }

   729→    let _idbPromise = null;

   731→    function openIdb() {
   732→      if (_idbPromise) return _idbPromise;
   733→
   734→      _idbPromise = new Promise((resolve, reject) => {
   735→        if (!('indexedDB' in window)) {
   736→          reject(new Error('IndexedDB недоступен'));
   737→          return;
   738→        }
   739→
   740→        const request = indexedDB.open(IDB_NAME, 1);
   741→
   742→        request.onupgradeneeded = () => {
   743→          const db = request.result;
   744→          if (!db.objectStoreNames.contains(IDB_STORE)) {
   745→            db.createObjectStore(IDB_STORE);
   746→          }
   747→        };
   748→
   749→        request.onsuccess = () => resolve(request.result);
   750→        request.onerror = () => reject(request.error);
   751→      });
   752→
   753→      _idbPromise.catch(() => {
   754→        _idbPromise = null;
   755→      });
   756→
   757→      return _idbPromise;
   758→    }

   760→    async function idbGet(key) {
   761→      try {
   762→        const db = await openIdb();
   763→
   764→        return await new Promise((resolve, reject) => {
   765→          const tx = db.transaction(IDB_STORE, 'readonly');
   766→          const req = tx.objectStore(IDB_STORE).get(key);
   767→
   768→          req.onsuccess = () => resolve(req.result);
   769→          req.onerror = () => reject(req.error);
   770→        });
   771→      } catch (error) {
   772→        console.warn('IndexedDB get error', error);
   773→        return undefined;
   774→      }
   775→    }

   777→    async function idbSet(key, value) {
   778→      try {
   779→        const db = await openIdb();

   781→        await new Promise((resolve, reject) => {
   782→          const tx = db.transaction(IDB_STORE, 'readwrite');
   783→          tx.objectStore(IDB_STORE).put(value, key);
   784→
   785→          tx.oncomplete = () => resolve();
   786→          tx.onerror = () => reject(tx.error);
   787→          tx.onabort = () => reject(tx.error);
   788→        });
   789→
   790→        return true;
   791→      } catch (error) {
   792→        console.warn('IndexedDB set error', error);

   817→    async function loadDocFromIdb(ydoc, key) {
   818→      const saved = await idbGet(key);
   819→
   820→      if (!saved) return;
   821→
   822→      try {
   823→        let u8;
   824→
   825→        if (saved instanceof Uint8Array) {
   826→          u8 = saved;
   827→        } else if (saved instanceof ArrayBuffer) {
   828→          u8 = new Uint8Array(saved);
   829→        } else if (Array.isArray(saved)) {
   830→          u8 = Uint8Array.from(saved);
   831→        } else {
   832→          return;
   833→        }
   834→
   835→        Y.applyUpdate(ydoc, u8, 'idb-load');
   836→      } catch (error) {
   837→        console.warn('Не удалось загрузить документ из IndexedDB', error);
   838→      }
   839→    }

   841→    function attachAutoSave(ydoc, key) {
   842→      let timer = 0;
   843→      let destroyed = false;
   844→
   845→      const save = async () => {
   846→        if (destroyed) return;
   847→
   848→        try {
   849→          const update = Y.encodeStateAsUpdate(ydoc);
   850→          await idbSet(key, update);
   851→        } catch (error) {
   852→          console.warn('AutoSave error', error);
   853→        }
   854→      };
   855→
   856→      const onUpdate = () => {
   857→        clearTimeout(timer);
   858→        timer = setTimeout(save, 400);
   859→      };
   860→
   861→      ydoc.on('update', onUpdate);
   862→
   863→      // Первая запись, чтобы создать пустой проект локально.
   864→      timer = setTimeout(save, 400);
   865→
   866→      return {
   867→        flush: save,
   868→        destroy() {
   869→          destroyed = true;
   870→          clearTimeout(timer);
   871→
   872→          try {
   873→            ydoc.off('update', onUpdate);
   874→          } catch {}
   875→        }
   876→      };
   877→    }

   889→    function yValueToPlain(value) {
   890→      if (value instanceof Y.Text) {
   891→        return value.toString();
   892→      }
   893→
   894→      if (value instanceof Y.Map) {
   895→        const result = {};
   896→        value.forEach((childValue, key) => {
   897→          result[key] = yValueToPlain(childValue);
   898→        });
   899→        return result;
   900→      }
   901→
   902→      if (value instanceof Y.Array) {
   903→        return value.toArray().map(item => yValueToPlain(item));
   904→      }
   905→
   906→      return value;
   907→    }

   909→    function plainToYValue(plain, key = undefined) {
   910→      if (plain === undefined) return undefined;
   911→      if (plain === null) return null;
   912→
   913→      if (typeof plain === 'string' && key && TEXT_KEYS.has(key)) {
   914→        return createYText(plain);
   915→      }
   916→
   917→      if (Array.isArray(plain)) {
   918→        const yArray = new Y.Array();
   919→        yArray.push(plain.map(item => plainToYValue(item)));
   920→        return yArray;
   921→      }
   922→
   923→      if (isPlainObject(plain)) {
   924→        if (plain.__yjs === 'Text') {
   925→          return createYText(plain.value || '');
   926→        }
   927→
   928→        const yMap = new Y.Map();
   929→        fillYMap(yMap, plain);
   930→        return yMap;
   931→      }
   932→
   933→      return plain;
   934→    }

   936→    function fillYMap(yMap, sourceObject) {
   937→      for (const [key, value] of Object.entries(sourceObject)) {
   938→        const yValue = plainToYValue(value, key);
   939→
   940→        if (yValue !== undefined) {
   941→          yMap.set(key, yValue);
   942→        }
   943→      }
   944→    }

   982→    function importProjectFromJson(json, ydoc, options = {}) {
   983→      const { clear = true } = options;
   984→
   985→      if (!json || json.format !== 'vetvi-project') {
   986→        throw new Error('Неверный формат проекта. Ожидался vetvi-project.');
   987→      }
   988→
   989→      if (json.schemaVersion > SCHEMA_VERSION) {
   990→        console.warn(`Версия схемы файла (${json.schemaVersion}) новее поддерживаемой (${SCHEMA_VERSION}).`);
   991→      }
   992→
   993→      ydoc.transact(() => {
   994→        if (clear) {
   995→          ROOTS.forEach(rootName => clearYMap(ydoc.getMap(rootName)));
   996→        }
   997→
   998→        for (const rootName of ROOTS) {
   999→          const source = json.data?.[rootName];
  1000→
  1001→          if (!source || !isPlainObject(source)) {
  1002→            continue;
  1003→          }
  1004→
  1005→          const targetMap = ydoc.getMap(rootName);
  1006→          fillYMap(targetMap, source);
  1007→        }
  1008→      }, 'import-project-json');
  1009→    }

  1011→    function patchYMap(targetMap, patchObject) {
  1012→      for (const [key, value] of Object.entries(patchObject)) {
  1013→        const existing = targetMap.get(key);
  1014→
  1015→        if (
  1016→          isPlainObject(value) &&
  1017→          existing instanceof Y.Map &&
  1018→          !TEXT_KEYS.has(key)
  1019→        ) {
  1020→          patchYMap(existing, value);
  1021→        } else {
  1022→          const yValue = plainToYValue(value, key);
  1023→
  1024→          if (yValue === undefined) {
  1025→            targetMap.delete(key);
  1026→          } else {
  1027→            targetMap.set(key, yValue);
  1028→          }
  1029→        }
  1030→      }
  1031→    }
