    67→function safeLocalStorage(): Storage | null {
    68→  try {
    69→    if (typeof window !== 'undefined' && window.localStorage) return window.localStorage

    72→  }

    76→function randomId(len = 24): string {
    77→  const alphabet = 'abcdefghijklmnopqrstuvwxyz0123456789'
    78→  const bytes = new Uint8Array(len)
    79→  if (typeof crypto !== 'undefined' && 'getRandomValues' in crypto) {
    80→    crypto.getRandomValues(bytes)
    81→  } else {
    82→    for (let i = 0; i < len; i++) bytes[i] = Math.floor(Math.random() * 256)
    83→  }
    84→  return Array.from(bytes)
    85→    .map((b) => alphabet[b % alphabet.length])
    86→    .join('')
    87→}

    89→function readHashParams(): URLSearchParams {
    90→  if (typeof window === 'undefined') return new URLSearchParams()
    91→  return new URLSearchParams(window.location.hash.replace(/^#/, ''))
    92→}
    93→
    94→function writeHashParams(paramsObject: Record<string, string | null>) {
    95→  if (typeof window === 'undefined') return
    96→  const params = readHashParams()
    97→  for (const [key, value] of Object.entries(paramsObject)) {
    98→    if (!value) params.delete(key)
    99→    else params.set(key, value)
   100→  }
   101→  const hash = params.toString()
   102→  const newUrl = `${window.location.pathname}${window.location.search}${hash ? `#${hash}` : ''}`
   103→  try {
   104→    window.history.replaceState(null, '', newUrl)
   105→  } catch {
   106→    window.location.hash = hash
   107→  }
   108→}

   169→  private notify(live = false) {

   190→  private buildSnapshot(): VetviSnapshot {
   191→    const doc = this.ydoc
   192→
   193→    let meta: ProjectMeta | null = null
   194→    const elements: ElementRecord[] = []
   195→
   196→    if (doc) {
   197→      const metaPlain = yValueToPlain(doc.getMap('meta')) as Record<string, unknown>
   198→      meta = {
   199→        projectId: String(metaPlain.projectId ?? this.projectId ?? ''),
   200→        name: String(metaPlain.name ?? ''),
   201→        schemaVersion: Number(metaPlain.schemaVersion ?? SCHEMA_VERSION),
   202→        createdAt: Number(metaPlain.createdAt ?? 0),
   203→        updatedAt: Number(metaPlain.updatedAt ?? 0),
   204→      }
   205→
   206→      doc.getMap('elements').forEach((value, id) => {
   207→        const rec = yValueToPlain(value) as Record<string, unknown>
   208→        elements.push({
   209→          id,
   210→          type: (rec.type as ElementType) ?? 'text',
   211→          x: Number(rec.x ?? 0),
   212→          y: Number(rec.y ?? 0),
   213→          width: Number(rec.width ?? 220),
   214→          height: Number(rec.height ?? 120),
   215→          text: String(rec.text ?? ''),
   216→          createdAt: Number(rec.createdAt ?? 0),
   217→          updatedAt: Number(rec.updatedAt ?? 0),
   218→        })
   219→      })
   220→
   221→      elements.sort(
   222→        (a, b) => (a.createdAt || 0) - (b.createdAt || 0) || a.id.localeCompare(b.id),
   223→      )
   224→    }
   225→
   226→    const peers: PeerInfo[] = []
   227→    const activeProvider = this.provider ?? this.mailboxProvider
   228→    if (activeProvider) {
   229→      activeProvider.awareness.getStates().forEach((state, clientId) => {
   230→        const user = (state as { user?: { name?: string; color?: string } } | undefined)?.user
   231→        peers.push({
   232→          clientId,
   233→          name: user?.name || `Участник ${String(clientId).slice(-4)}`,
   234→          color: user?.color || '#9aa0a6',
   235→          isSelf: clientId === activeProvider.awareness.clientID,
   236→        })
   237→      })
   238→      peers.sort((a, b) => (a.isSelf ? -1 : b.isSelf ? 1 : a.clientId - b.clientId))
   239→    }
   240→
   241→    const connection = this.mode === 'copy' ? this.mailboxConnection : this.connection
   242→
   243→    return {
   244→      ready: this.ready,
   245→      mode: this.mode,
   246→      meta,
   247→      elements,
   248→      selectedId: this.selectedId,
   249→      connection,
   250→      roomActive: this.provider !== null,
   251→      mailboxActive: this.mailboxProvider !== null,
   252→      peers,
   253→      peersCount: peers.length,
   254→      copies: this.copies,
   255→      incoming: this.mode === 'project' ? this.readIncoming() : [],
   256→      outgoing: this.outbox.filter((r) =>
   257→        this.copyMeta ? r.copyId === this.copyMeta.copyId : false,
   258→      ),
   259→      backups: this.backups,
   260→      shareUrl: this.shareUrl,
   261→      roomId: this.roomId,
   262→      copyMeta: this.copyMeta,
   263→      saveState: this.saveState,
   264→      userName: this.userName,
   265→      userColor: this.userColor,
   266→    }
   267→  }

   269→  private readIncoming(): MergeRequest[] {
   270→    if (!this.mailboxDoc) return []
   271→    const requestsMap = this.mailboxDoc.getMap('requests')
   272→    const requests: MergeRequest[] = []
   273→    requestsMap.forEach((value, id) => {
   274→      const plain = yValueToPlain(value) as Record<string, unknown> | null
   275→      if (!plain) return
   276→      requests.push({
   277→        id: String(plain.id ?? id),
   278→        senderId: String(plain.senderId ?? ''),
   279→        senderName: String(plain.senderName ?? 'Неизвестный'),
   280→        copyId: String(plain.copyId ?? ''),
   281→        sourceRoomId: plain.sourceRoomId ? String(plain.sourceRoomId) : null,
   282→        createdAt: Number(plain.createdAt ?? 0),
   283→        delta: (plain.delta ?? {}) as MergeRequest['delta'],
   284→        stats: (plain.stats ?? { added: 0, updated: 0, deleted: 0 }) as MergeRequest['stats'],
   285→        status: (plain.status as MergeRequest['status']) ?? 'pending',
   286→      })
   287→    })
   288→    requests.sort((a, b) => b.createdAt - a.createdAt)
   289→    return requests
   290→  }

   296→  async bootstrap() {
   297→    if (this.bootstrapped || typeof window === 'undefined') return
   298→    this.bootstrapped = true
   299→
   300→    const storage = safeLocalStorage()
   301→    this.userId = storage?.getItem(LS_USER_ID) || uid()
   302→    storage?.setItem(LS_USER_ID, this.userId)
   303→
   304→    this.userName = storage?.getItem(LS_USER_NAME) || `Гость-${this.userId.slice(0, 4)}`
   305→    storage?.setItem(LS_USER_NAME, this.userName)
   306→
   307→    let colorIndex = 0
   308→    for (let i = 0; i < this.userId.length; i++) colorIndex += this.userId.charCodeAt(i)
   309→    this.userColor = USER_COLORS[colorIndex % USER_COLORS.length]
   310→
   311→    await this.routeFromHash(readHashParams())
   312→
   313→    this.ready = true
   314→
   315→    window.addEventListener('hashchange', () => {
   316→      void this.handleHashChange()
   317→    })
   318→
   319→    this.rebuildAndEmit()
   320→  }

   322→  private async handleHashChange() {
   323→    const params = readHashParams()
   324→    const room = params.get('room')
   325→    const copyId = params.get('copy')
   326→
   327→    if (room && room !== this.roomId) {
   328→      await this.routeFromHash(params)
   329→      return
   330→    }
   331→    if (copyId && this.copyMeta?.copyId !== copyId) {
   332→      await this.routeFromHash(params)
   333→      return
   334→    }
   335→    if (!room && !copyId && (this.roomId || this.copyMeta)) {
   336→      await this.routeFromHash(params)
   337→    }
   338→  }
   339→
   340→  private async routeFromHash(params: URLSearchParams) {
   341→    const room = params.get('room')
   342→    const project = params.get('project')
   343→    const copyId = params.get('copy')
   344→
   345→    if (copyId) {
   346→      const record = await getCopy(copyId)
   347→      if (record) {
   348→        await this.openCopy(copyId)
   349→        return
   350→      }
   351→    }
   352→
   353→    if (room) {
   354→      const projectId = project || uid()
   355→      await this.openProject(projectId, { name: project ? 'Проект' : 'Проект из комнаты' })
   356→      this.connectRoom(room)
   357→      writeHashParams({ room, project: projectId, copy: null })
   358→      return
   359→    }
   360→
   361→    // Без параметров — последний локальный проект или новый.
   362→    const storage = safeLocalStorage()
   363→    const last = storage?.getItem(LS_LAST_PROJECT)
   364→    if (last && (await getProjectUpdate(last))) {
   365→      await this.openProject(last)
   366→      return
   367→    }
   368→
   369→    await this.newProject({ silent: true })
   370→  }

   389→  private async createDoc(projectId: string, name: string) {
   390→    const doc = new Y.Doc()
   391→    this.ydoc = doc
   392→    this.projectId = projectId
   393→
   394→    const saved = await getProjectUpdate(projectId)
   395→    if (saved) {
   396→      try {
   397→        Y.applyUpdate(doc, saved, 'idb-load')
   398→      } catch (error) {
   399→        console.warn('Не удалось загрузить проект из IndexedDB', error)
   400→      }
   401→    }
   402→
   403→    ensureProjectSkeleton(doc, projectId, name)
   404→    await addProjectId(projectId)
   405→
   406→    doc.on('update', () => this.notify())
   407→    this.scheduleSave()
   408→  }

   419→  private cleanupDoc() {
   420→    this.disconnectRoom()
   421→
   422→    if (this.autoSaveTimer) {
   423→      clearTimeout(this.autoSaveTimer)
   424→      this.autoSaveTimer = null
   425→    }
   426→    if (this.textTimer) {
   427→      clearTimeout(this.textTimer)
   428→      this.textTimer = null
   429→    }
   430→
   431→    try {
   432→      this.ydoc?.destroy()
   433→    } catch {
   434→      // ignore
   435→    }
   436→    this.ydoc = null
   437→    this.selectedId = null
   438→    this.saveState = 'saved'
   439→  }
   440→
   441→  private scheduleSave() {
   442→    if (this.autoSaveTimer) clearTimeout(this.autoSaveTimer)
   443→    this.saveState = 'saving'
   444→    this.autoSaveTimer = setTimeout(async () => {
   445→      this.autoSaveTimer = null
   446→      const doc = this.ydoc
   447→      if (!doc) return
   448→      try {
   449→        const update = Y.encodeStateAsUpdate(doc)
   450→        if (this.mode === 'copy' && this.currentCopyRecord) {
   451→          await saveCopy({
   452→            ...this.currentCopyRecord,
   453→            snapshotUpdate: uint8ToBase64(update),
   454→            meta: { ...this.currentCopyRecord.meta, lastOpenedAt: Date.now() },
   455→          })
   456→        } else if (this.projectId) {
   457→          await saveProjectUpdate(this.projectId, update)
   458→        }
   459→        this.saveState = 'saved'
   460→        this.notify()
   461→      } catch (error) {
   462→        console.warn('AutoSave error', error)
   463→        this.saveState = 'saved'
   464→      }
   465→    }, 400)
   466→  }

   497→  addElement(type: ElementType, position?: { x: number; y: number }) {
   498→    const doc = this.ydoc
   499→    if (!doc) return
   500→    const now = Date.now()
   501→    const id = uid()
   502→    const count = doc.getMap('elements').size
   503→    const fallback = { x: 140 + (count % 6) * 36, y: 120 + (count % 6) * 32 }
   504→
   505→    upsertRecord(doc, 'elements', {
   506→      id,
   507→      type,
   508→      x: position?.x ?? fallback.x,
   509→      y: position?.y ?? fallback.y,
   510→      width: 220,
   511→      height: 120,
   512→      text: '',
   513→      createdAt: now,
   514→      updatedAt: now,
   515→    })
   516→
   517→    this.selectedId = id
   518→    this.notify(true)
   519→  }

   521→  updateElement(id: string, patch: Partial<Omit<ElementRecord, 'id'>>, live = false) {
   522→    const doc = this.ydoc
   523→    if (!doc) return
   524→    const existingValue = doc.getMap('elements').get(id)
   525→    if (!(existingValue instanceof Y.Map)) return
   526→    const existing = yValueToPlain(existingValue) as Record<string, unknown>
   527→
   528→    upsertRecord(doc, 'elements', {
   529→      ...existing,
   530→      ...patch,
   531→      id,
   532→      updatedAt: Date.now(),
   533→    })
   534→    this.scheduleSave()
   535→    this.notify(live)
   536→  }
   537→
   538→  deleteElement(id: string) {
   539→    const doc = this.ydoc
   540→    if (!doc) return
   541→    deleteRecord(doc, 'elements', id)
   542→    if (this.selectedId === id) this.selectedId = null
   543→    this.scheduleSave()
   544→    this.notify(true)
   545→  }

   547→  selectElement(id: string | null) {
   548→    if (this.selectedId === id) return
   549→    this.selectedId = id
   550→    this.notify(true)

   557→  async share(): Promise<string | null> {
   558→    if (this.mode === 'copy') {
   559→      this.toast('Это локальная копия — поделиться ссылкой нельзя', 'warn')
   560→      return null
   561→    }
   562→    if (!this.ydoc || !this.projectId) return null
   563→
   564→    if (this.roomId && this.provider) {
   565→      this.copyShareLink()
   566→      return this.shareUrl
   567→    }
   568→
   569→    const room = randomId(24)
   570→    this.connectRoom(room)
   571→    writeHashParams({ room, project: this.projectId, copy: null })
   572→    this.toast('Комната открыта — отправьте ссылку соавтору')
   573→    this.notify(true)
   574→    return this.shareUrl
   575→  }

   577→  private connectRoom(room: string) {
   578→    this.disconnectRoom()
   579→    if (!this.ydoc) return
   580→
   581→    this.roomId = room
   582→
   583→    this.provider = new VetviProvider(room, this.ydoc, {
   584→      onStatus: (status) => {
   585→        this.connection = status
   586→        this.notify(true)
   587→      },
   588→      onPeerJoined: () => {
   589→        this.toast('Участник подключился к проекту')
   590→        this.notify(true)
   591→      },
   592→      onPeerLeft: () => {
   593→        this.notify(true)
   594→      },
   595→      onAwareness: () => this.notify(true),
   596→      onJoinAck: () => {
   597→        this.notify(true)
   598→      },
   599→    })
   600→
   601→    this.provider.awareness.setLocalStateField('user', {
   602→      name: this.userName,
   603→      color: this.userColor,
   604→    })
   605→
   606→    this.shareUrl = this.buildShareUrl(room, this.projectId ?? '')
   607→
   608→    // Владелец/участник слушает merge-mailbox оригинала.
   609→    this.connectMailbox(`${room}::mailbox`)
   610→  }
   611→
   612→  disconnectRoom() {
   613→    if (this.provider) {
   614→      try {
   615→        this.provider.destroy()
   616→      } catch {
   617→        // ignore
   618→      }
   619→      this.provider = null
   620→    }
   621→    this.disconnectMailbox()
   622→    this.roomId = null

   839→  async sendMerge(): Promise<boolean> {
   840→    const doc = this.ydoc
   841→    if (!doc || this.mode !== 'copy' || !this.copyMeta) {
   842→      this.toast('Merge доступен только из копии', 'warn')
   843→      return false
   844→    }
   845→
   846→    if (!this.copyMeta.sourceRoomId) {
   847→      this.toast('У копии нет комнаты оригинала — используйте экспорт JSON', 'error')
   848→      return false
   849→    }
   850→
   851→    try {
   852→      const current = exportProjectToJson(doc).data as unknown as CopyRecord['baseState']
   853→      const base = this.currentCopyRecord?.baseState ?? ({} as CopyRecord['baseState'])
   854→      const delta = computeDelta(this.copyMeta.sourceProjectId, base, current)
   855→
   856→      if (isDeltaEmpty(delta)) {
   857→        this.toast('Нет изменений для merge — копия совпадает с базой', 'warn')
   858→        return false
   859→      }
   860→
   861→      const request: MergeRequest = {
   862→        id: uid(),
   863→        senderId: this.userId,
   864→        senderName: this.userName,
   865→        copyId: this.copyMeta.copyId,
   866→        sourceRoomId: this.copyMeta.sourceRoomId,
   867→        createdAt: Date.now(),
   868→        delta,
   869→        stats: deltaStats(delta),
   870→        status: this.mailboxProvider ? 'sent' : 'queued',
   871→      }
   872→
   873→      if (this.mailboxProvider && this.mailboxDoc) {
   874→        this.mailboxDoc.transact(() => {
   875→          this.mailboxDoc!.getMap('requests').set(request.id, request)
   876→        }, 'merge-send')
   877→      }
   878→
   879→      this.outbox = [request, ...this.outbox.filter((r) => r.id !== request.id)]
   880→      this.persistOutbox()
   881→
   882→      if (request.status === 'queued') {
   883→        this.toast('Владелец офлайн — merge-запрос в очереди, отправим при подключении', 'warn')
   884→      } else {
   885→        this.toast(`Merge-запрос отправлен (+${request.stats.added} / ~${request.stats.updated} / −${request.stats.deleted})`)
   886→      }
   887→      this.notify(true)
   888→      return true
   889→    } catch (error) {
   890→      console.error('sendMerge failed', error)
   891→      this.toast('Не удалось отправить merge-запрос', 'error')
   892→      return false
   893→    }
   894→  }
   895→
   896→  private async retryOutbox() {
   897→    const queued = this.outbox.filter((r) => r.status === 'queued')
   898→    if (!queued.length) return
   899→    if (!this.mailboxProvider || !this.mailboxDoc) return
   900→
   901→    for (const request of queued) {
   902→      try {
   903→        this.mailboxDoc.transact(() => {
   904→          this.mailboxDoc!.getMap('requests').set(request.id, request)
   905→        }, 'merge-retry')
   906→        request.status = 'sent'
   907→      } catch (error) {
   908→        console.warn('retryOutbox failed', error)
   909→        request.status = 'failed'
   910→      }
   911→    }
   912→    this.persistOutbox()
   913→    if (queued.some((r) => r.status === 'sent')) {
   914→      this.toast('Отложенные merge-запросы отправлены')
   915→    }
   916→    this.notify(true)
   917→  }
   918→
   919→  private syncOutboxStatuses() {
   920→    if (!this.mailboxDoc) return
   921→    const requestsMap = this.mailboxDoc.getMap('requests')
   922→    let changed = false
   923→
   924→    for (const item of this.outbox) {
   925→      const remote = requestsMap.get(item.id)
   926→      if (!remote) continue
   927→      const plain = yValueToPlain(remote) as Record<string, unknown>
   928→      const status = plain.status as MergeRequest['status']
   929→      if (status && status !== item.status && (status === 'applied' || status === 'rejected')) {
   930→        item.status = status
   931→        changed = true
   932→        if (status === 'applied') {
   933→          this.toast('Владелец принял ваш merge-запрос')
   934→        } else {
   935→          this.toast('Владелец отклонил ваш merge-запрос', 'warn')
   936→        }
   937→      }
   938→    }
   939→
   940→    if (changed) this.persistOutbox()
   941→  }

   943→  async acceptMerge(requestId: string) {
   944→    if (!this.ydoc || !this.mailboxDoc) {
   945→      this.toast('Merge-комната не подключена', 'error')
   946→      return
   947→    }
   948→
   949→    const remote = this.mailboxDoc.getMap('requests').get(requestId)
   950→    if (!remote) {
   951→      this.toast('Merge-запрос не найден', 'error')
   952→      return
   953→    }
   954→
   955→    const plain = yValueToPlain(remote) as Record<string, unknown>
   956→    const delta = (plain.delta ?? {}) as MergeRequest['delta']
   957→    const senderName = String(plain.senderName ?? 'Неизвестный')
   958→
   959→    try {
   960→      // FR-11: backup перед применением.
   961→      const backupId = uid()
   962→      const backupMeta: BackupMeta = {
   963→        id: backupId,
   964→        projectId: this.projectId ?? 'unknown',
   965→        createdAt: Date.now(),
   966→        label: `Перед merge от ${senderName}`,
   967→      }
   968→      await saveBackup(backupMeta, Y.encodeStateAsUpdate(this.ydoc!))
   969→      await this.refreshBackups()
   970→
   971→      applyDelta(this.ydoc!, delta)
   972→      this.scheduleSave()
   973→
   974→      this.mailboxDoc.transact(() => {
   975→        const existing = this.mailboxDoc!.getMap('requests').get(requestId)
   976→        if (existing instanceof Y.Map) {
   977→          existing.set('status', 'applied')
   978→        } else {
   979→          this.mailboxDoc!.getMap('requests').set(requestId, { ...plain, status: 'applied' })
   980→        }
   981→      }, 'merge-accept')
   982→
   983→      this.toast(`Merge от ${senderName} применён (создан backup)`)
   984→      this.notify(true)
   985→    } catch (error) {
   986→      console.error('acceptMerge failed', error)
   987→      this.toast('Ошибка применения merge-запроса', 'error')
   988→    }
   989→  }

   991→  async rejectMerge(requestId: string) {

   994→      const plain = yValueToPlain(this.mailboxDoc.getMap('requests').get(requestId)) as
   995→        | Record<string, unknown>

   997→      if (!plain) {

  1000→      }

  1002→        const existing = this.mailboxDoc!.getMap('requests').get(requestId)
  1003→        if (existing instanceof Y.Map) {
  1004→          existing.set('status', 'rejected')
  1005→        } else {
  1006→          this.mailboxDoc!.getMap('requests').set(requestId, { ...plain, status: 'rejected' })
  1007→        }
  1008→      }, 'merge-reject')
  1009→      this.toast('Merge-запрос отклонён')
  1010→      this.notify(true)

  1015→  }

  1094→    }

  1096→      lines.push('=== Текстовые блоки ===')
  1097→      texts.forEach((e) => lines.push(e.text))
  1098→      lines.push('')
  1099→    }
  1100→
  1101→    if (!snap.elements.length) lines.push('(проект пуст)')
