    76→    76→function randomId(len = 24): string {
    77→    77→  const alphabet = 'abcdefghijklmnopqrstuvwxyz0123456789'
    78→    78→  const bytes = new Uint8Array(len)
    79→    79→  if (typeof crypto !== 'undefined' && 'getRandomValues' in crypto) {
    80→    80→    crypto.getRandomValues(bytes)
    81→    81→  } else {
    82→    82→    for (let i = 0; i < len; i++) bytes[i] = Math.floor(Math.random() * 256)
    83→    83→  }
    84→    84→  return Array.from(bytes)
    85→    85→    .map((b) => alphabet[b % alphabet.length])
    86→    86→    .join('')
    87→    87→}

    89→    89→function readHashParams(): URLSearchParams {
    90→    90→  if (typeof window === 'undefined') return new URLSearchParams()
    91→    91→  return new URLSearchParams(window.location.hash.replace(/^#/, ''))
    92→    92→}
    93→    93→
    94→    94→function writeHashParams(paramsObject: Record<string, string | null>) {
    95→    95→  if (typeof window === 'undefined') return
    96→    96→  const params = readHashParams()
    97→    97→  for (const [key, value] of Object.entries(paramsObject)) {
    98→    98→    if (!value) params.delete(key)
    99→    99→    else params.set(key, value)
   100→   100→  }
   101→   101→  const hash = params.toString()
   102→   102→  const newUrl = `${window.location.pathname}${window.location.search}${hash ? `#${hash}` : ''}`
   103→   103→  try {
   104→   104→    window.history.replaceState(null, '', newUrl)
   105→   105→  } catch {
   106→   106→    window.location.hash = hash
   107→   107→  }
   108→   108→}

   190→   190→  private buildSnapshot(): VetviSnapshot {
   191→   191→    const doc = this.ydoc
   192→   192→
   193→   193→    let meta: ProjectMeta | null = null
   194→   194→    const elements: ElementRecord[] = []
   195→   195→
   196→   196→    if (doc) {
   197→   197→      const metaPlain = yValueToPlain(doc.getMap('meta')) as Record<string, unknown>
   198→   198→      meta = {
   199→   199→        projectId: String(metaPlain.projectId ?? this.projectId ?? ''),
   200→   200→        name: String(metaPlain.name ?? ''),
   201→   201→        schemaVersion: Number(metaPlain.schemaVersion ?? SCHEMA_VERSION),
   202→   202→        createdAt: Number(metaPlain.createdAt ?? 0),
   203→   203→        updatedAt: Number(metaPlain.updatedAt ?? 0),
   204→   204→      }
   205→   205→
   206→   206→      doc.getMap('elements').forEach((value, id) => {
   207→   207→        const rec = yValueToPlain(value) as Record<string, unknown>
   208→   208→        elements.push({
   209→   209→          id,
   210→   210→          type: (rec.type as ElementType) ?? 'text',
   211→   211→          x: Number(rec.x ?? 0),
   212→   212→          y: Number(rec.y ?? 0),
   213→   213→          width: Number(rec.width ?? 220),
   214→   214→          height: Number(rec.height ?? 120),
   215→   215→          text: String(rec.text ?? ''),
   216→   216→          createdAt: Number(rec.createdAt ?? 0),
   217→   217→          updatedAt: Number(rec.updatedAt ?? 0),
   218→   218→        })
   219→   219→      })
   220→   220→
   221→   221→      elements.sort(
   222→   222→        (a, b) => (a.createdAt || 0) - (b.createdAt || 0) || a.id.localeCompare(b.id),
   223→   223→      )
   224→   224→    }
   225→   225→
   226→   226→    const peers: PeerInfo[] = []
   227→   227→    const activeProvider = this.provider ?? this.mailboxProvider
   228→   228→    if (activeProvider) {
   229→   229→      activeProvider.awareness.getStates().forEach((state, clientId) => {
   230→   230→        const user = (state as { user?: { name?: string; color?: string } } | undefined)?.user
   231→   231→        peers.push({
   232→   232→          clientId,
   233→   233→          name: user?.name || `Участник ${String(clientId).slice(-4)}`,
   234→   234→          color: user?.color || '#9aa0a6',
   235→   235→          isSelf: clientId === activeProvider.awareness.clientID,
   236→   236→        })
   237→   237→      })
   238→   238→      peers.sort((a, b) => (a.isSelf ? -1 : b.isSelf ? 1 : a.clientId - b.clientId))
   239→   239→    }
   240→   240→
   241→   241→    const connection = this.mode === 'copy' ? this.mailboxConnection : this.connection
   242→   242→
   243→   243→    return {
   244→   244→      ready: this.ready,
   245→   245→      mode: this.mode,
   246→   246→      meta,
   247→   247→      elements,
   248→   248→      selectedId: this.selectedId,
   249→   249→      connection,
   250→   250→      roomActive: this.provider !== null,
   251→   251→      mailboxActive: this.mailboxProvider !== null,
   252→   252→      peers,
   253→   253→      peersCount: peers.length,
   254→   254→      copies: this.copies,
   255→   255→      incoming: this.mode === 'project' ? this.readIncoming() : [],
   256→   256→      outgoing: this.outbox.filter((r) =>
   257→   257→        this.copyMeta ? r.copyId === this.copyMeta.copyId : false,
   258→   258→      ),
   259→   259→      backups: this.backups,
   260→   260→      shareUrl: this.shareUrl,
   261→   261→      roomId: this.roomId,
   262→   262→      copyMeta: this.copyMeta,
   263→   263→      saveState: this.saveState,
   264→   264→      userName: this.userName,
   265→   265→      userColor: this.userColor,
   266→   266→    }
   267→   267→  }

   322→   322→  private async handleHashChange() {
   323→   323→    const params = readHashParams()
   324→   324→    const room = params.get('room')
   325→   325→    const copyId = params.get('copy')
   326→   326→
   327→   327→    if (room && room !== this.roomId) {
   328→   328→      await this.routeFromHash(params)
   329→   329→      return
   330→   330→    }
   331→   331→    if (copyId && this.copyMeta?.copyId !== copyId) {
   332→   332→      await this.routeFromHash(params)
   333→   333→      return
   334→   334→    }
   335→   335→    if (!room && !copyId && (this.roomId || this.copyMeta)) {
   336→   336→      await this.routeFromHash(params)
   337→   337→    }
   338→   338→  }

   340→   340→  private async routeFromHash(params: URLSearchParams) {
   341→   341→    const room = params.get('room')
   342→   342→    const project = params.get('project')
   343→   343→    const copyId = params.get('copy')
   344→   344→
   345→   345→    if (copyId) {
   346→   346→      const record = await getCopy(copyId)
   347→   347→      if (record) {
   348→   348→        await this.openCopy(copyId)
   349→   349→        return
   350→   350→      }
   351→   351→    }
   352→   352→
   353→   353→    if (room) {
   354→   354→      const projectId = project || uid()
   355→   355→      await this.openProject(projectId, { name: project ? 'Проект' : 'Проект из комнаты' })
   356→   356→      this.connectRoom(room)
   357→   357→      writeHashParams({ room, project: projectId, copy: null })
   358→   358→      return
   359→   359→    }
   360→   360→
   361→   361→    // Без параметров — последний локальный проект или новый.
   362→   362→    const storage = safeLocalStorage()
   363→   363→    const last = storage?.getItem(LS_LAST_PROJECT)
   364→   364→    if (last && (await getProjectUpdate(last))) {
   365→   365→      await this.openProject(last)
   366→   366→      return
   367→   367→    }
   368→   368→
   369→   369→    await this.newProject({ silent: true })
   370→   370→  }

   441→   441→  private scheduleSave() {
   442→   442→    if (this.autoSaveTimer) clearTimeout(this.autoSaveTimer)
   443→   443→    this.saveState = 'saving'
   444→   444→    this.autoSaveTimer = setTimeout(async () => {
   445→   445→      this.autoSaveTimer = null
   446→   446→      const doc = this.ydoc
   447→   447→      if (!doc) return
   448→   448→      try {
   449→   449→        const update = Y.encodeStateAsUpdate(doc)
   450→   450→        if (this.mode === 'copy' && this.currentCopyRecord) {
   451→   451→          await saveCopy({
   452→   452→            ...this.currentCopyRecord,
   453→   453→            snapshotUpdate: uint8ToBase64(update),
   454→   454→            meta: { ...this.currentCopyRecord.meta, lastOpenedAt: Date.now() },
   455→   455→          })
   456→   456→        } else if (this.projectId) {
   457→   457→          await saveProjectUpdate(this.projectId, update)
   458→   458→        }
   459→   459→        this.saveState = 'saved'
   460→   460→        this.notify()
   461→   461→      } catch (error) {
   462→   462→        console.warn('AutoSave error', error)
   463→   463→        this.saveState = 'saved'
   464→   464→      }
   465→   465→    }, 400)
   466→   466→  }

   497→   497→  addElement(type: ElementType, position?: { x: number; y: number }) {
   498→   498→    const doc = this.ydoc

   839→   839→  async sendMerge(): Promise<boolean> {
   840→   840→    const doc = this.ydoc
   841→   841→    if (!doc || this.mode !== 'copy' || !this.copyMeta) {
   842→   842→      this.toast('Merge доступен только из копии', 'warn')
   843→   843→      return false
   844→   844→    }
   845→   845→
   846→   846→    if (!this.copyMeta.sourceRoomId) {
   847→   847→      this.toast('У копии нет комнаты оригинала — используйте экспорт JSON', 'error')
   848→   848→      return false
   849→   849→    }
   850→   850→
   851→   851→    try {
   852→   852→      const current = exportProjectToJson(doc).data as unknown as CopyRecord['baseState']
   853→   853→      const base = this.currentCopyRecord?.baseState ?? ({} as CopyRecord['baseState'])
   854→   854→      const delta = computeDelta(this.copyMeta.sourceProjectId, base, current)
   855→   855→
   856→   856→      if (isDeltaEmpty(delta)) {
   857→   857→        this.toast('Нет изменений для merge — копия совпадает с базой', 'warn')
   858→   858→        return false
   859→   859→      }
   860→   860→
   861→   861→      const request: MergeRequest = {
   862→   862→        id: uid(),
   863→   863→        senderId: this.userId,
   864→   864→        senderName: this.userName,
   865→   865→        copyId: this.copyMeta.copyId,
   866→   866→        sourceRoomId: this.copyMeta.sourceRoomId,
   867→   867→        createdAt: Date.now(),
   868→   868→        delta,
   869→   869→        stats: deltaStats(delta),
   870→   870→        status: this.mailboxProvider ? 'sent' : 'queued',
   871→   871→      }
   872→   872→
   873→   873→      if (this.mailboxProvider && this.mailboxDoc) {
   874→   874→        this.mailboxDoc.transact(() => {
   875→   875→          this.mailboxDoc!.getMap('requests').set(request.id, request)
   876→   876→        }, 'merge-send')
   877→   877→      }
   878→   878→
   879→   879→      this.outbox = [request, ...this.outbox.filter((r) => r.id !== request.id)]
   880→   880→      this.persistOutbox()
   881→   881→
   882→   882→      if (request.status === 'queued') {
   883→   883→        this.toast('Владелец офлайн — merge-запрос в очереди, отправим при подключении', 'warn')
   884→   884→      } else {
   885→   885→        this.toast(`Merge-запрос отправлен (+${request.stats.added} / ~${request.stats.updated} / −${request.stats.deleted})`)
   886→   886→      }
   887→   887→      this.notify(true)
   888→   888→      return true
   889→   889→    } catch (error) {
   890→   890→      console.error('sendMerge failed', error)
   891→   891→      this.toast('Не удалось отправить merge-запрос', 'error')
   892→   892→      return false
   893→   893→    }
   894→   894→  }
   895→   895→
   896→   896→  private async retryOutbox() {
   897→   897→    const queued = this.outbox.filter((r) => r.status === 'queued')
   898→   898→    if (!queued.length) return
   899→   899→    if (!this.mailboxProvider || !this.mailboxDoc) return
   900→   900→
   901→   901→    for (const request of queued) {
   902→   902→      try {
   903→   903→        this.mailboxDoc.transact(() => {
   904→   904→          this.mailboxDoc!.getMap('requests').set(request.id, request)
   905→   905→        }, 'merge-retry')
   906→   906→        request.status = 'sent'
   907→   907→      } catch (error) {
   908→   908→        console.warn('retryOutbox failed', error)
   909→   909→        request.status = 'failed'
   910→   910→      }
   911→   911→    }
   912→   912→    this.persistOutbox()
   913→   913→    if (queued.some((r) => r.status === 'sent')) {
   914→   914→      this.toast('Отложенные merge-запросы отправлены')
   915→   915→    }
   916→   916→    this.notify(true)
   917→   917→  }
   918→   918→
   919→   919→  private syncOutboxStatuses() {
   920→   920→    if (!this.mailboxDoc) return
   921→   921→    const requestsMap = this.mailboxDoc.getMap('requests')
   922→   922→    let changed = false
   923→   923→
   924→   924→    for (const item of this.outbox) {
   925→   925→      const remote = requestsMap.get(item.id)
   926→   926→      if (!remote) continue
   927→   927→      const plain = yValueToPlain(remote) as Record<string, unknown>
   928→   928→      const status = plain.status as MergeRequest['status']
   929→   929→      if (status && status !== item.status && (status === 'applied' || status === 'rejected')) {
   930→   930→        item.status = status
   931→   931→        changed = true
   932→   932→        if (status === 'applied') {
   933→   933→          this.toast('Владелец принял ваш merge-запрос')
   934→   934→        } else {
   935→   935→          this.toast('Владелец отклонил ваш merge-запрос', 'warn')
   936→   936→        }
   937→   937→      }
   938→   938→    }
   939→   939→
   940→   940→    if (changed) this.persistOutbox()
   941→   941→  }

   943→   943→  async acceptMerge(requestId: string) {
   944→   944→    if (!this.ydoc || !this.mailboxDoc) {
   945→   945→      this.toast('Merge-комната не подключена', 'error')
   946→   946→      return
   947→   947→    }
   948→   948→
   949→   949→    const remote = this.mailboxDoc.getMap('requests').get(requestId)
   950→   950→    if (!remote) {
   951→   951→      this.toast('Merge-запрос не найден', 'error')
   952→   952→      return
   953→   953→    }
   954→   954→
   955→   955→    const plain = yValueToPlain(remote) as Record<string, unknown>
   956→   956→    const delta = (plain.delta ?? {}) as MergeRequest['delta']
   957→   957→    const senderName = String(plain.senderName ?? 'Неизвестный')
   958→   958→
   959→   959→    try {
   960→   960→      // FR-11: backup перед применением.
   961→   961→      const backupId = uid()
   962→   962→      const backupMeta: BackupMeta = {
   963→   963→        id: backupId,
   964→   964→        projectId: this.projectId ?? 'unknown',
   965→   965→        createdAt: Date.now(),
   966→   966→        label: `Перед merge от ${senderName}`,
   967→   967→      }
   968→   968→      await saveBackup(backupMeta, Y.encodeStateAsUpdate(this.ydoc!))
   969→   969→      await this.refreshBackups()
   970→   970→
   971→   971→      applyDelta(this.ydoc!, delta)
   972→   972→      this.scheduleSave()
   973→   973→
   974→   974→      this.mailboxDoc.transact(() => {
   975→   975→        const existing = this.mailboxDoc!.getMap('requests').get(requestId)
   976→   976→        if (existing instanceof Y.Map) {
   977→   977→          existing.set('status', 'applied')
   978→   978→        } else {
   979→   979→          this.mailboxDoc!.getMap('requests').set(requestId, { ...plain, status: 'applied' })
   980→   980→        }
   981→   981→      }, 'merge-accept')
   982→   982→
   983→   983→      this.toast(`Merge от ${senderName} применён (создан backup)`)
