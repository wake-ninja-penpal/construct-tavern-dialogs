    76→    76→    76→function randomId(len = 24): string {
    77→    77→    77→  const alphabet = 'abcdefghijklmnopqrstuvwxyz0123456789'
    78→    78→    78→  const bytes = new Uint8Array(len)
    79→    79→    79→  if (typeof crypto !== 'undefined' && 'getRandomValues' in crypto) {
    80→    80→    80→    crypto.getRandomValues(bytes)
    81→    81→    81→  } else {
    82→    82→    82→    for (let i = 0; i < len; i++) bytes[i] = Math.floor(Math.random() * 256)
    83→    83→    83→  }
    84→    84→    84→  return Array.from(bytes)
    85→    85→    85→    .map((b) => alphabet[b % alphabet.length])
    86→    86→    86→    .join('')
    87→    87→    87→}

    94→    94→    94→function writeHashParams(paramsObject: Record<string, string | null>) {
    95→    95→    95→  if (typeof window === 'undefined') return
    96→    96→    96→  const params = readHashParams()
    97→    97→    97→  for (const [key, value] of Object.entries(paramsObject)) {
    98→    98→    98→    if (!value) params.delete(key)
    99→    99→    99→    else params.set(key, value)
   100→   100→   100→  }
   101→   101→   101→  const hash = params.toString()
   102→   102→   102→  const newUrl = `${window.location.pathname}${window.location.search}${hash ? `#${hash}` : ''}`
   103→   103→   103→  try {
   104→   104→   104→    window.history.replaceState(null, '', newUrl)
   105→   105→   105→  } catch {
   106→   106→   106→    window.location.hash = hash
   107→   107→   107→  }
   108→   108→   108→}

   334→   334→   334→    }

   336→   336→   336→      await this.routeFromHash(params)
